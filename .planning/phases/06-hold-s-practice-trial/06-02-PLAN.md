**2. Add the `holdKeyPracticeBlock()` factory function** (place it after `sKeyInstructionTrial` and before `tappingInstructionsTimeline`):

```typescript
/**
 * Creates a hold-key practice block for Phase 6.
 * Participant holds S key for ~5 seconds, releases on prompt.
 * Loop exits after 2 successes OR 3 failures (whichever first).
 * "Entraînement réussi" screen shown after loop exits.
 *
 * IMPORTANT: Uses jsPsych.data.filter() for counting (not closure variables)
 * so that counters survive the outer "Repeat Practice" loop in buildPracticeTrials().
 */
const holdKeyPracticeBlock = (
  jsPsych: JsPsych,
  state: ExperimentState,
): Trial => ({
  timeline: [
    {
      // Inner looping node: repeats the hold-key trial
      timeline: [
        {
          type: HoldKeyPracticePlugin,
          holdKey() {
            return getHoldKeys(state)[0];
          },
          holdDuration: HOLD_KEY_PRACTICE_DURATION,
          data: {
            task: 'hold-key-practice',
          },
        },
      ],
      loop_function() {
        const practiceTrials = jsPsych.data
          .get()
          .filter({ task: 'hold-key-practice' })
          .values();
        const successes = practiceTrials.filter(
          (t: Record<string, unknown>) => t.success === true,
        ).length;
        const failures = practiceTrials.filter(
          (t: Record<string, unknown>) => t.success === false,
        ).length;
        // Continue looping while BOTH thresholds are unmet
        return (
          successes < HOLD_KEY_MIN_SUCCESSES &&
          failures < HOLD_KEY_MAX_FAILURES
        );
      },
    },
    // End screen — OUTSIDE the looping node (shown once after loop exits)
    {
      type: HtmlButtonResponsePlugin,
      stimulus: () => HOLD_S_PRACTICE_COMPLETE_MESSAGE(),
      choices: [CONTINUE_BUTTON_MESSAGE()],
    },
  ],
});
```
